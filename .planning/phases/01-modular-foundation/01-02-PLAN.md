```powershell
function Get-KitRoot { [CmdletBinding()] [OutputType([string])] param() }            # kit/ folder (parent of kit/core)
function Get-KitPackagePath {
    [CmdletBinding()] [OutputType([string])]
    param(
        [Parameter(Mandatory)][string]$OutputRoot,
        [Parameter(Mandatory)][ValidatePattern('^[A-Za-z0-9][A-Za-z0-9._-]*$')][string]$Vendor,
        [Parameter(Mandatory)][ValidatePattern('^[A-Za-z0-9][A-Za-z0-9._-]*$')][string]$ModelId,
        [Parameter(Mandatory)][ValidatePattern('^[A-Za-z0-9][A-Za-z0-9._-]*$')][string]$Os,
        [Parameter(Mandatory)][ValidatePattern('^[A-Za-z0-9][A-Za-z0-9._-]*$')][string]$OsVer
    )
}   # -> <OutputRoot>\<vendor lowercased>\<ModelId>\<Os>-<OsVer>
function ConvertTo-KitCommandLine { [CmdletBinding()] [OutputType([string])] param([Parameter(Mandatory)][AllowEmptyCollection()][AllowEmptyString()][string[]]$ArgumentList) }
function Invoke-KitTool {
    [CmdletBinding(SupportsShouldProcess)]
    param([Parameter(Mandatory, Position = 0)][string]$FilePath, [Parameter(Position = 1)][string[]]$ArgumentList = @(), [int[]]$OkCode = @(0, 3010))
}
```

Existing escaping logic to move into ConvertTo-KitCommandLine (src/Install-HPDriverBaseline.ps1 lines 60-72): empty string becomes `""`; a value without whitespace or quotes passes unchanged; otherwise backslashes immediately before a quote are doubled and the quote escaped (`(\\*)"` -> `$1$1\"`), trailing backslashes are doubled (`(\\+)$` -> `$1$1`), and the value is wrapped in double quotes; values are joined with single spaces.
