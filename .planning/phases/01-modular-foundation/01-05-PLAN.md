```powershell
function Get-KitProfile {           # parameter sets ByModel (default) and ByPath
    [CmdletBinding(DefaultParameterSetName = 'ByModel')] [OutputType([hashtable])]
    param([Parameter(Mandatory, ParameterSetName = 'ByModel')][string]$Vendor,
          [Parameter(ParameterSetName = 'ByModel')][string]$ModelId,
          [Parameter(Mandatory, ParameterSetName = 'ByPath')][string]$Path,
          [string]$ProfileRoot = (Join-Path (Get-KitRoot) 'profiles'))
}   # -> the profile hashtable plus a ProfilePath key; without -ModelId, every profile of the vendor sorted by ModelId
function Get-KitVerifiedValue {
    [CmdletBinding()]
    param([Parameter(Mandatory)][hashtable]$ModelProfile, [Parameter(Mandatory)][ValidateSet('FirmwareMinimum', 'BiosDefaults', 'Platform')][string]$Name, [switch]$AllowUnverified)
}   # FirmwareMinimum -> @{ Version; SoftPaq }, BiosDefaults -> Settings hashtable, Platform -> string
function Get-KitConfig {
    [CmdletBinding()] [OutputType([pscustomobject])]
    param([string]$Root = (Get-KitRoot), [hashtable]$Override = @{})
}   # -> one property per example section (ThirdParty = hashtable), plus Source ('Local'|'Defaults') and LocalPath
```
