Core functions this plan creates (names already listed in Kit.Core.psd1 by Plan 01-02):
```powershell
function Get-KitVendor {            # parameter sets: All (default), ByName (-Name), ByManufacturer (-Manufacturer), ByPath (-DescriptorPath)
    [CmdletBinding(DefaultParameterSetName = 'All')] [OutputType([pscustomobject])]
    param([string]$Name, [string]$Manufacturer, [string]$DescriptorPath, [string[]]$VendorRoot = @(Join-Path (Get-KitRoot) 'vendors'))
}   # -> PSTypeName 'Kit.Vendor': Name, Manufacturer, Folder, DescriptorPath, TargetModulePath, BuildModulePath (or $null when the file is absent), Commands
function Test-KitVendorContract {
    [CmdletBinding()] [OutputType([pscustomobject])]
    param([Parameter(Mandatory)][string]$DescriptorPath, [ValidateSet('Build', 'Target')][string[]]$Plane = @('Build', 'Target'))
}   # -> zero or more PSTypeName 'Kit.ContractViolation': Vendor, Operation, Kind, Detail
    #    Kind is one of: InvalidDescriptor, UnknownOperation, MissingOperation, ModuleNotFound, CommandNotFound, MissingParameter
function Invoke-KitVendorCommand {
    [CmdletBinding()]
    param([Parameter(Mandatory)][pscustomobject]$Vendor,
          [Parameter(Mandatory)][ValidateSet('ListModels','GetVendorPack','NewCustomPack','GetFirmware','InstallDrivers','InstallFirmware','GetDeviceIdentity','GetInstalledFirmwareVersion')][string]$Operation,
          [hashtable]$Parameters = @{})
}
```
